find_package(Boost 1.92 CONFIG REQUIRED COMPONENTS redis)
find_package(OpenSSL REQUIRED)

add_library(jobq_redis_transport STATIC RedisTransport.cpp)
target_include_directories(jobq_redis_transport PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_features(jobq_redis_transport PUBLIC cxx_std_20)
target_link_libraries(jobq_redis_transport PRIVATE Boost::redis OpenSSL::SSL OpenSSL::Crypto Threads::Threads)
if(WIN32)
    target_link_libraries(jobq_redis_transport PRIVATE ws2_32 mswsock crypt32)
endif()

if(BUILD_TESTING)
    add_executable(jobq_redis_test "${PROJECT_SOURCE_DIR}/tests/backend/redis/RedisTest.cpp")
    target_link_libraries(jobq_redis_test PRIVATE jobq_redis_transport)
    add_test(NAME jobq_redis_transport COMMAND jobq_redis_test)
    add_test(NAME jobq_redis_live COMMAND jobq_redis_test --live)
    set_tests_properties(jobq_redis_transport jobq_redis_live PROPERTIES TIMEOUT 30)
    set_tests_properties(jobq_redis_live PROPERTIES SKIP_RETURN_CODE 77)
endif()
